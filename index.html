var banks = {};
function initBanks(){
  try{
    if(typeof allPools_ur !== 'undefined') banks['ur']=allPools_ur;
    if(typeof allPools_en !== 'undefined') banks['en']=allPools_en;
    if(typeof allPools_ar !== 'undefined') banks['ar']=allPools_ar;
    if(typeof allPools_tr !== 'undefined') banks['tr']=allPools_tr;
    if(typeof allPools_fr !== 'undefined') banks['fr']=allPools_fr;
    if(typeof allPools_de !== 'undefined') banks['de']=allPools_de;
    if(typeof allPools_es !== 'undefined') banks['es']=allPools_es;
    if(typeof allPools_bn !== 'undefined') banks['bn']=allPools_bn;
    if(typeof allPools_hi !== 'undefined') banks['hi']=allPools_hi;
    if(typeof allPools_id !== 'undefined') banks['id']=allPools_id;
    // Fallback to ur if any missing
    for(var l in banks){
      if(!banks[l]) banks[l]=banks['ur'];
    }
    console.log('Banks ready:', Object.keys(banks).length);
  }catch(e){ console.error('Banks init error', e); }
}
function getQuestionBank(lang){
  if(Object.keys(banks).length===0) initBanks();
  var b=banks[lang];
  if(!b){ console.warn('No bank for', lang, 'using ur'); b=banks['ur']; }
  return b;
}
function buildAllLevels(lang){
  initBanks();
  var bank=getQuestionBank(lang);
  if(!bank){ console.error('No bank at all'); return []; }
  var keys=['quran','hadith','anbiya','namaz','zakat','roza','hajj','seerat','sahaba','jannat'];
  var all=[];
  for(var L=0; L<100; L++){
    var lvl=[];
    for(var i=0;i<keys.length;i++){
      var k=keys[i];
      var pool=bank[k];
      if(pool && pool[L]) lvl.push(pool[L]);
      else if(pool && pool[0]) lvl.push(pool[0]);
      else lvl.push({q:"Question",o:["1","2","3"],a:"1",cat:k});
    }
    all.push(lvl);
  }
  console.log('Built', all.length, 'levels for', lang, 'first q:', all[0][0].q);
  return all;
}
initBanks();
